# When building with Vulkan under the AGP/NDK toolchain, the shader compiler
# (glslc) is not part of the CMake install — but the NDK ships it in
# shader-tools/<host-tag>. FindVulkan (CMake >= 3.24) discovers glslc through
# a VULKAN_SDK-style layout, so provide one shim directory in the build tree.
# AGP passes the NDK as ANDROID_NDK; the toolchain file also exposes
# CMAKE_ANDROID_NDK — accept either.
set(_LOCALAI_NDK "${ANDROID_NDK}")
if(NOT _LOCALAI_NDK)
    set(_LOCALAI_NDK "${CMAKE_ANDROID_NDK}")
endif()
if(LLAMA_VULKAN AND _LOCALAI_NDK)
    find_program(GLSLC_PROBE NAMES glslc
        PATHS "$ENV{VULKAN_SDK}/bin"
        NO_DEFAULT_PATH)
    if(NOT GLSLC_PROBE)
        foreach(_host linux-x86_64 darwin-x86_64 windows-x86_64)
            if(EXISTS "${_LOCALAI_NDK}/shader-tools/${_host}/glslc")
                set(_shim "${CMAKE_BINARY_DIR}/vulkan-shim")
                file(MAKE_DIRECTORY "${_shim}/bin")
                set(_real_glslc "${_LOCALAI_NDK}/shader-tools/${_host}/glslc")
                # NDK's glslc resolves libc++.so from its own directory (rpath);
                # the wrapper below execs the real binary in place, so the
                # library never moves.
                if(EXISTS "${_LOCALAI_NDK}/shader-tools/${_host}/libc++.so")
                    file(COPY_FILE
                        "${_LOCALAI_NDK}/shader-tools/${_host}/libc++.so"
                        "${_shim}/bin/libc++.so")
                endif()
                # ggml-vulkan detects shader-extension support by matching the
                # phrase "extension not supported: <ext>" against glslc stderr
                # with a CMake regex whose dot cannot cross newlines. Old NDK
                # glslc prints the phrase late in a multi-line report, so the
                # check misfires and later shader compiles fail. The wrapper
                # repeats the phrase on stderr's first line, making the
                # detection exact for every glslc version.
                file(WRITE "${_shim}/bin/glslc"
                    "#!/bin/sh\n"
                    "# Generated by app/src/main/cpp/CMakeLists.txt — do not edit.\n"
                    "err_tmp=\$(mktemp \"\${TMPDIR:-/tmp}/glslc-err.XXXXXX\")\n"
                    "\"${_real_glslc}\" \"\$@\" 2>\"\$err_tmp\"\n"
                    "rc=\$?\n"
                    "if [ -s \"\$err_tmp\" ] && grep -q \"extension not supported\" \"\$err_tmp\"; then\n"
                    "    grep -m1 \"extension not supported\" \"\$err_tmp\" >&2\n"
                    "fi\n"
                    "cat \"\$err_tmp\" >&2\n"
                    "rm -f \"\$err_tmp\"\n"
                    "exit \$rc\n")
                file(CHMOD "${_shim}/bin/glslc"
                    PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE
                                GROUP_READ GROUP_EXECUTE
                                WORLD_READ WORLD_EXECUTE)
                # CMake >= 3.31 hard-requires the glslangValidator component even
                # when only glslc is requested. ggml-vulkan never invokes it — the
                # shaders compile through glslc exclusively — so a documented stub
                # keeps the completeness check happy without a Vulkan SDK install.
                file(WRITE "${_shim}/bin/glslangValidator"
                    "#!/bin/sh\n"
                    "# Stub: the NDK ships no glslangValidator and ggml-vulkan does\n"
                    "# not use it (shaders compile via glslc). Satisfies the CMake\n"
                    "# FindVulkan component check only.\n"
                    "echo 'glslangValidator stub: not used by this build' >&2\n"
                    "exit 1\n")
                file(CHMOD "${_shim}/bin/glslangValidator"
                    PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE
                                GROUP_READ GROUP_EXECUTE
                                WORLD_READ WORLD_EXECUTE)
                list(PREPEND CMAKE_PREFIX_PATH "${_shim}")
                # The NDK toolchain sets FIND_ROOT_PATH_MODE_PACKAGE=ONLY, which
                # would ignore our prefix in find_package; name the package dir
                # directly and root the shim explicitly.
                set(SPIRV-Headers_DIR "${_shim}/lib/cmake/spirv-headers" CACHE PATH "SPIRV-Headers config (generated)" FORCE)
                list(PREPEND CMAKE_FIND_ROOT_PATH "${_shim}")
                set(ENV{VULKAN_SDK} "${_shim}")
                # ggml-vulkan also hard-requires the SPIRV-Headers CONFIG package.
                # The headers themselves are fetched by scripts/fetch_deps.sh;
                # generate the minimal package config that points at them.
                set(_spirv_include "${CMAKE_CURRENT_SOURCE_DIR}/spirv-headers/include")
                if(EXISTS "${_spirv_include}/spirv/unified1/spirv.hpp")
                    file(MAKE_DIRECTORY "${_shim}/lib/cmake/spirv-headers")
                    file(WRITE "${_shim}/lib/cmake/spirv-headers/SPIRV-HeadersConfig.cmake"
                        "# Generated by app/src/main/cpp/CMakeLists.txt — do not edit.\n"
                        "set(SPIRV-Headers_INCLUDE_DIRS \"${_spirv_include}\")\n"
                        "if(NOT TARGET SPIRV-Headers)\n"
                        "    add_library(SPIRV-Headers INTERFACE)\n"
                        "    target_include_directories(SPIRV-Headers INTERFACE \"${_spirv_include}\")\n"
                        "endif()\n")
                    message(STATUS "Vulkan: SPIRV-Headers package config from ${_spirv_include}")
                else()
                    message(WARNING "Vulkan: SPIRV-Headers sources missing — run scripts/fetch_deps.sh")
                endif()
                unset(_spirv_include)
                message(STATUS "Vulkan: using NDK glslc via shim ${_shim}")
                break()
            endif()
        endforeach()
    endif()
    unset(GLSLC_PROBE CACHE)
endif()
unset(_LOCALAI_NDK)

# ggml-vulkan compiles against the C++ bindings (vulkan/vulkan.hpp) that the
# NDK sysroot does not carry; scripts/fetch_deps.sh pins them beside llama.cpp.
# SPIRV headers are included directly from ggml-vulkan-types.h, so both include
# roots are added at directory level before the llama.cpp subdirectory.
if(LLAMA_VULKAN)
    set(_vulkan_hpp "${CMAKE_CURRENT_SOURCE_DIR}/vulkan-headers/include/vulkan/vulkan.hpp")
    set(_spirv_hpp "${CMAKE_CURRENT_SOURCE_DIR}/spirv-headers/include/spirv/unified1/spirv.hpp")
    if(EXISTS "${_vulkan_hpp}")
        include_directories("${CMAKE_CURRENT_SOURCE_DIR}/vulkan-headers/include")
    else()
        message(WARNING "Vulkan: vulkan.hpp missing — run scripts/fetch_deps.sh")
    endif()
    if(EXISTS "${_spirv_hpp}")
        include_directories("${CMAKE_CURRENT_SOURCE_DIR}/spirv-headers/include")
    else()
        message(WARNING "Vulkan: SPIRV headers missing — run scripts/fetch_deps.sh")
    endif()
    unset(_vulkan_hpp)
    unset(_spirv_hpp)
endif()

add_subdirectory("${LLAMA_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama.cpp" EXCLUDE_FROM_ALL)

add_library(localai SHARED llama_jni.cpp)
target_include_directories(localai PRIVATE "${LLAMA_DIR}/include")
target_link_libraries(localai PRIVATE llama android log)
target_compile_options(localai PRIVATE
    -O3
    -fexceptions
    -frtti
    -Wall
    -Wextra
    -Wno-unused-parameter
    -Wno-unused-function)
